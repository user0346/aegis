# AEGIS V2 — Voice Extras (optional, fuer freihaendiges "Hey Jarvis" + Sprache)
# Installation:
#   pip install -r requirements_v2_voice.txt
#
# Alles LOKAL/OFFLINE, kein API-Key. Audio-Capture ueber sounddevice (hat Wheels inkl.
# PortAudio auch fuer Python 3.14 — pyaudio/webrtcvad bauen dort NICHT). pyaudio bleibt
# optionaler Fallback fuer aeltere Pythons.

# ── Spracheingabe: Aufnahme + Erkennung (STT) ───────────────────────────────
sounddevice>=0.5            # Mikro-Aufnahme (PortAudio gebundelt; 3.14-tauglich)
faster-whisper>=1.2         # lokale Spracherkennung (offline, kein Key)

# ── GPU-Beschleunigung der Spracherkennung (NVIDIA) ─────────────────────────
# Mit diesen Libs laeuft faster-whisper auf der Grafikkarte (schnell + genau, deutsch-
# spezialisiertes Modell). FEHLEN sie, nutzt AEGIS automatisch die CPU (langsamer, aber
# funktioniert) — stt.py macht die DLLs auffindbar und faellt sauber zurueck. Nur fuer
# NVIDIA-GPUs sinnvoll; auf anderen Systemen werden sie einfach nicht geladen.
nvidia-cublas-cu12; platform_system == 'Windows'
nvidia-cudnn-cu12; platform_system == 'Windows'
nvidia-cuda-runtime-cu12; platform_system == 'Windows'
nvidia-cuda-nvrtc-cu12; platform_system == 'Windows'

# ── Weckwort "Hey Jarvis" (freihaendig, offline) ────────────────────────────
openwakeword>=0.6           # Apache-2.0-Code; vortrainiertes 'hey_jarvis'-Modell
onnxruntime>=1.26.0           # Inferenz-Backend fuer openWakeWord (Windows: onnx)

# ── Optionaler Fallback (aeltere Pythons, z.B. 3.13) ────────────────────────
PyAudio>=0.2.14            # Audio-Capture-Fallback (braucht ggf. PortAudio/MSVC)
webrtcvad-wheels>=2.0.14   # VAD fuer den pyaudio-Pfad (pre-built wheels)
